# GSP751 – Interact with Terraform Modules

## Known lab bugs (read first)

1. Checkpoints need two checks each. "Provision infrastructure" gives 25 points for creating the VPC and 25 for destroying it. Click Check my progress after apply, then destroy, then click it again. The lab never says this, so most students get stuck at 25/50.
2. Don't run the cleanup before checking. The lab puts `terraform destroy` and `rm -rd terraform-google-network` before the Check my progress button. If you delete the folder first, the editor keeps showing deleted files (tab names struck through) and the next apply fails with "Reference to undeclared input variable network_name".
3. The error.html download URL is broken. The lab uses `raw.githubusercontent.com/.../blob/master/...`, and `/blob/` doesn't work on raw URLs, so you download a 404 page. Fixed below.
4. The root output is quoted. The lab writes `value = "module.gcs-static-website-bucket.bucket"`, which outputs that literal text instead of the bucket. It doesn't break the check, but it's wrong. Fixed below.
5. Region and subnet names can differ per student. Use the values from your lab panel in the Setup block.

## Setup (edit to match your lab panel)

Students only change the value right after echo on the first four lines, and only if their lab instructions show something different:

```bash
echo us-central1 > ~/region
echo subnet-01 > ~/subnet1
echo subnet-02 > ~/subnet2
echo subnet-03 > ~/subnet3
gcloud config get-value project > ~/project
cat ~/region ~/subnet1 ~/subnet2 ~/subnet3 ~/project
```

## Task 1 – Use a module from the Registry

```bash
cd ~
rm -rf terraform-google-network
git clone https://github.com/terraform-google-modules/terraform-google-network
cd terraform-google-network
git checkout tags/v6.0.1 -b v6.0.1
cd examples/simple_project

cat > variables.tf <<EOF
variable "project_id" {
  description = "The project ID to host the network in"
  default     = "`cat ~/project`"
}

variable "network_name" {
  description = "The name of the network to be created"
  default     = "example-vpc"
}
EOF

cat > main.tf <<EOF
module "test-vpc-module" {
  source       = "terraform-google-modules/network/google"
  version      = "~> 6.0"
  project_id   = var.project_id
  network_name = var.network_name
  mtu          = 1460

  subnets = [
    {
      subnet_name   = "`cat ~/subnet1`"
      subnet_ip     = "10.10.10.0/24"
      subnet_region = "`cat ~/region`"
    },
    {
      subnet_name           = "`cat ~/subnet2`"
      subnet_ip             = "10.10.20.0/24"
      subnet_region         = "`cat ~/region`"
      subnet_private_access = "true"
      subnet_flow_logs      = "true"
    },
    {
      subnet_name               = "`cat ~/subnet3`"
      subnet_ip                 = "10.10.30.0/24"
      subnet_region             = "`cat ~/region`"
      subnet_flow_logs          = "true"
      subnet_flow_logs_interval = "INTERVAL_10_MIN"
      subnet_flow_logs_sampling = 0.7
      subnet_flow_logs_metadata = "INCLUDE_ALL_METADATA"
      subnet_flow_logs_filter   = "false"
    }
  ]
}
EOF

terraform init
terraform apply -auto-approve
```

Check my progress: Provision infrastructure (25/50)

```bash
cd ~/terraform-google-network/examples/simple_project
terraform destroy -auto-approve
```

Check my progress: Provision infrastructure again (50/50)

## Task 2 – Build a module (module files)

```bash
cd ~
rm -rf terraform-google-network
mkdir -p ~/modules/gcs-static-website-bucket
cd ~/modules/gcs-static-website-bucket

cat > website.tf <<'EOF'
resource "google_storage_bucket" "bucket" {
  name                        = var.name
  project                     = var.project_id
  location                    = var.location
  storage_class               = var.storage_class
  labels                      = var.labels
  force_destroy               = var.force_destroy
  uniform_bucket_level_access = true

  versioning {
    enabled = var.versioning
  }

  dynamic "retention_policy" {
    for_each = var.retention_policy == null ? [] : [var.retention_policy]
    content {
      is_locked        = var.retention_policy.is_locked
      retention_period = var.retention_policy.retention_period
    }
  }

  dynamic "encryption" {
    for_each = var.encryption == null ? [] : [var.encryption]
    content {
      default_kms_key_name = var.encryption.default_kms_key_name
    }
  }

  dynamic "lifecycle_rule" {
    for_each = var.lifecycle_rules
    content {
      action {
        type          = lifecycle_rule.value.action.type
        storage_class = lookup(lifecycle_rule.value.action, "storage_class", null)
      }
      condition {
        age                   = lookup(lifecycle_rule.value.condition, "age", null)
        created_before        = lookup(lifecycle_rule.value.condition, "created_before", null)
        with_state            = lookup(lifecycle_rule.value.condition, "with_state", null)
        matches_storage_class = lookup(lifecycle_rule.value.condition, "matches_storage_class", null)
        num_newer_versions    = lookup(lifecycle_rule.value.condition, "num_newer_versions", null)
      }
    }
  }
}
EOF

cat > variables.tf <<'EOF'
variable "name" {
  type = string
}
variable "project_id" {
  type = string
}
variable "location" {
  type = string
}
variable "storage_class" {
  type    = string
  default = null
}
variable "labels" {
  type    = map(string)
  default = null
}
variable "versioning" {
  type    = bool
  default = true
}
variable "force_destroy" {
  type    = bool
  default = true
}
variable "retention_policy" {
  type = object({
    is_locked        = bool
    retention_period = number
  })
  default = null
}
variable "encryption" {
  type = object({
    default_kms_key_name = string
  })
  default = null
}
variable "lifecycle_rules" {
  type = list(object({
    action    = any
    condition = any
  }))
  default = []
}
EOF

cat > outputs.tf <<'EOF'
output "bucket" {
  description = "The created storage bucket"
  value       = google_storage_bucket.bucket
}
EOF

echo "# GCS static website bucket" > README.md
echo "Licensed under the Apache License, Version 2.0" > LICENSE
```

## Task 2 – Root module, apply, upload

```bash
cd ~

cat > main.tf <<EOF
module "gcs-static-website-bucket" {
  source = "./modules/gcs-static-website-bucket"

  name       = var.name
  project_id = var.project_id
  location   = "`cat ~/region`"

  lifecycle_rules = [{
    action = {
      type = "Delete"
    }
    condition = {
      age        = 365
      with_state = "ANY"
    }
  }]
}
EOF

cat > outputs.tf <<'EOF'
output "bucket-name" {
  description = "Bucket names."
  value       = module.gcs-static-website-bucket.bucket.name
}
EOF

cat > variables.tf <<EOF
variable "project_id" {
  type    = string
  default = "`cat ~/project`"
}
variable "name" {
  type    = string
  default = "`cat ~/project`"
}
EOF

terraform init
terraform apply -auto-approve

curl -s https://raw.githubusercontent.com/hashicorp/learn-terraform-modules/master/modules/aws-s3-static-website-bucket/www/index.html > index.html
curl -s https://raw.githubusercontent.com/hashicorp/learn-terraform-modules/master/modules/aws-s3-static-website-bucket/www/error.html > error.html
gcloud storage cp index.html error.html gs://`cat ~/project`
gcloud storage ls gs://`cat ~/project`
```

Check my progress: Upload files to the bucket

If it shows 25/50, destroy and check again:

```bash
cd ~
terraform destroy -auto-approve
```

Check my progress: Upload files to the bucket (50/50)
